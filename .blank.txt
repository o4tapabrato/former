export async function PATCH(
    request: Request,
    { params }: { params: { id: string } }
) {
    try {
        const userId = await getCurrentUser();
        if (!userId) {
            return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
        }

        const body = await request.json();

        // Verify ownership first
        const existingSurvey = await db.survey.findUnique({
            where: { id: params.id },
        });

        if (!existingSurvey || existingSurvey.userId !== userId) {
            return NextResponse.json({ error: "Survey not found or unauthorized" }, { status: 404 });
        }

        // Optional safety check: Ensure it's still unpublished (draft) if you want to restrict editing published ones
        // if (existingSurvey.published) {
        //     return NextResponse.json({ error: "Cannot edit a published survey" }, { status: 400 });
        // }

        // Update survey details and reset/update questions cleanly
        const updatedSurvey = await db.$transaction(async (tx) => {
            // Delete old questions and recreate them, or update them. Recreating is clean for builders:
            await tx.surveyQuestion.deleteMany({
                where: { surveyId: params.id },
            });

            return await tx.survey.update({
                where: { id: params.id },
                data: {
                    title: body.title,
                    description: body.description,
                    expiresAt: body.expiresAt ? new Date(body.expiresAt) : null,
                    published: body.published ?? false,
                    questions: {
                        create: (body.questions || []).map((q: any, index: number) => ({
                            type: q.type,
                            text: q.text,
                            order: q.order ?? index,
                            required: q.required || false,
                            options: q.options || [],
                            imageUrl: q.imageUrl || null,
                            maxRating: q.maxRating || null,
                        })),
                    },
                },
                include: { questions: true },
            });
        });

        return NextResponse.json({ success: true, survey: updatedSurvey }, { status: 200 });
    } catch (error) {
        console.error("Failed to update survey:", error);
        return NextResponse.json({ error: "Internal Server Error" }, { status: 500 });
    }
}
